// 7. 本地极速过滤及重新渲染列表
function filterAndRender() {
  const query = state.currentQuery.toLowerCase();
  
  const filtered = fallbackDatabase.filter(item => {
    const matchQuery = item.title.toLowerCase().includes(query) || 
                       item.snippet.toLowerCase().includes(query);
    const matchCategory = state.category === "all" || item.type === state.category;
    return matchQuery && matchCategory;
  });

  renderResults(filtered);
}

// 8. 绘制结果列表卡片
function renderResults(data) {
  resultsColumn.innerHTML = "";
  
  if (data.length === 0) {
    resultsColumn.innerHTML = `
      <div style="text-align: center; color: var(--text-muted); padding: 50px 20px; font-size: 13.5px;">
        <i data-lucide="info" style="width: 22px; height: 22px; margin: 0 auto 10px auto; display: block; stroke: var(--text-dim);"></i>
        <div>没有在本地数据库中找到关联记录</div>
      </div>
    `;
    lucide.createIcons();
    return;
  }

  data.forEach(item => {
    const card = document.createElement('div');
    const isSelected = state.activeCardId === item.id;
    
    card.className = `result-card ${isSelected ? 'selected' : ''} ${state.engineMode === 'ahk' ? 'ahk-theme' : ''}`;
    card.setAttribute('data-id', item.id);
    
    // 根据类型分配合适的极简图标
    let iconName = "file-text";
    if (item.type === "code") iconName = "code-2";
    if (item.type === "web") iconName = "globe";

    card.innerHTML = `
      <div class="rc-header">
        <div class="rc-title-container">
          <i data-lucide="${iconName}" style="width:15px; height:15px; color: var(--text-muted); flex-shrink: 0;"></i>
          <span class="rc-title">${item.title}</span>
        </div>
        <span class="rc-meta">${item.type.toUpperCase()}</span>
      </div>
      <p class="rc-snippet">${item.snippet}</p>
      <div class="rc-footer">
        <div class="rc-footer-left">
          <span class="rc-footer-item">
            <i data-lucide="folder" style="width:12px; height:12px; color: var(--text-dim);"></i>
            ${item.path}
          </span>
        </div>
      </div>
    `;

    card.addEventListener('click', () => selectCard(item, card));
    resultsColumn.appendChild(card);
  });
  
  lucide.createIcons();
}

// 9. 精确选中卡片 + 骨架屏转场效果 (AI 体验灵魂所在)
function selectCard(item, cardElement) {
  // 清除全部卡片选中状态
  document.querySelectorAll('.result-card').forEach(c => {
    c.classList.remove('selected');
  });
  
  cardElement.classList.add('selected');
  state.activeCardId = item.id;

  // 1. 隐藏默认 Placeholder 状态
  previewPlaceholder.style.opacity = '0';
  setTimeout(() => {
    previewPlaceholder.style.display = 'none';
  }, 200);

  // 2. 隐藏真实面板，打开丝滑骨架屏
  previewWrapper.classList.remove('active');
  skeletonLoader.style.display = 'flex';

  // 模拟极速底层 IPC 分发过渡 (300毫秒，既保证科技感又不耽误使用)
  setTimeout(() => {
    prevTitle.textContent = item.title;
    prevPath.textContent = item.path;
    prevType.textContent = item.type.toUpperCase();
    prevDesc.textContent = item.desc;
    
    if (item.code) {
      prevCode.textContent = item.code;
      codeBlockContainer.style.display = 'block';
    } else {
      codeBlockContainer.style.display = 'none';
    }

    // 3. 关闭骨架，呈现结果
    skeletonLoader.style.display = 'none';
    previewWrapper.classList.add('active');
  }, 300);

  // 通知底层 AHK 当前选择变更
  postToAhk({ type: "selectItem", id: item.id, title: item.title });
}

// 10. 重置预览视图
function resetPreview() {
  previewWrapper.classList.remove('active');
  skeletonLoader.style.display = 'none';
  previewPlaceholder.style.display = 'flex';
  setTimeout(() => {
    previewPlaceholder.style.opacity = '1';
  }, 50);
  state.activeCardId = null;
  document.querySelectorAll('.result-card').forEach(c => c.classList.remove('selected'));
}

// 11. 支持来自 AHK 底层的实时消息注入接口 (不破环原本的双向绑定逻辑)
window.addEventListener("message", (event) => {
  const payload = event.data;
  if (!payload) return;
  
  // AHK 底层数据库更新时重载
  if (payload.type === "updateDatabase") {
    // 支持实时覆盖
    if (payload.data && Array.isArray(payload.data)) {
      fallbackDatabase.length = 0;
      fallbackDatabase.push(...payload.data);
      filterAndRender();
    }
  }
  
  // 处理外部同步状态上报
  if (payload.type === "syncStatus") {
    syncStatusText.textContent = payload.text || "已完成全索引同步";
    if (payload.syncing) {
      syncStatusDot.className = "dock-status-dot syncing";
    } else {
      syncStatusDot.className = "dock-status-dot";
    }
  }
});
